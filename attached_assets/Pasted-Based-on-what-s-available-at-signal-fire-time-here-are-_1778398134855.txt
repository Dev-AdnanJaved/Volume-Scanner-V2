pythondef calculate_monster_score(signal):
    ad = signal['additional_data']
    score = 0

    if ad.get('funding_in_ideal_range') == True:
        score += 1

    if ad.get('is_compressed') == False:
        score += 1

    mcap = ad.get('market_cap_usd') or 0
    if 5_000_000 <= mcap <= 200_000_000:
        score += 1

    if (signal.get('quality_score') or 0) >= 4:
        score += 1

    if (signal.get('soft_flags') or 0) <= 2:
        score += 1

    fr = ad.get('funding_rate') or 0
    if fr > -0.01:
        score += 1

    if fr > 0:
        score += 1

    if ad.get('price_above_ema50_4h') == True:
        score += 1

    if signal.get('btc_trend_at_entry') == 'ranging':
        score += 1

    if signal.get('signal_type') in ['fast', 'slow']:
        score += 1

    return score  # 0 to 10
